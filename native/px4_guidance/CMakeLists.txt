cmake_minimum_required(VERSION 3.16)
project(pixeagle_px4_guidance VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
include(CTest)
option(PIXEAGLE_BUILD_PX4_MODE "Build the pinned ROS2 external-mode bridge" OFF)
add_library(command_lease INTERFACE)
target_include_directories(command_lease INTERFACE include)
find_package(nlohmann_json REQUIRED)
add_library(local_protocol STATIC src/local_server.cpp)
target_link_libraries(local_protocol PUBLIC command_lease nlohmann_json::nlohmann_json)
if(PIXEAGLE_BUILD_PX4_MODE)
    set(PIXEAGLE_EXPECTED_PX4_REVISION "" CACHE STRING "Exact firmware baseline from dependencies.lock.json")
    string(LENGTH "${PIXEAGLE_EXPECTED_PX4_REVISION}" revision_length)
    if(NOT revision_length EQUAL 40 OR NOT PIXEAGLE_EXPECTED_PX4_REVISION MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "Native mode requires the pinned 40-character PIXEAGLE_EXPECTED_PX4_REVISION")
    endif()
    string(SUBSTRING "${PIXEAGLE_EXPECTED_PX4_REVISION}" 0 16 firmware_hash)
    find_package(rclcpp REQUIRED)
    find_package(px4_ros2_cpp REQUIRED)
    find_package(px4_msgs REQUIRED)
    find_package(eigen3_cmake_module REQUIRED)
    find_package(Eigen3 REQUIRED)
    add_library(pixeagle_external_mode STATIC src/external_mode.cpp src/aircraft_observations.cpp)
    target_link_libraries(pixeagle_external_mode PUBLIC command_lease rclcpp::rclcpp
        px4_ros2_cpp::px4_ros2_cpp Eigen3::Eigen nlohmann_json::nlohmann_json)
    add_executable(pixeagle-native-guidance src/main.cpp)
    target_compile_definitions(pixeagle-native-guidance PRIVATE PIXEAGLE_EXPECTED_FIRMWARE_HASH="${firmware_hash}")
    target_link_libraries(pixeagle-native-guidance PRIVATE pixeagle_external_mode local_protocol)
    install(TARGETS pixeagle-native-guidance DESTINATION bin)
endif()
if(BUILD_TESTING)
    find_package(GTest REQUIRED)
    add_executable(command_lease_test test/command_lease_test.cpp)
    target_link_libraries(command_lease_test PRIVATE command_lease GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(command_lease_test)
    add_executable(local_protocol_test test/local_protocol_test.cpp)
    target_link_libraries(local_protocol_test PRIVATE local_protocol GTest::gtest_main)
    gtest_discover_tests(local_protocol_test)
    add_executable(source_sample_test test/source_sample_test.cpp)
    target_link_libraries(source_sample_test PRIVATE command_lease GTest::gtest_main)
    gtest_discover_tests(source_sample_test)
    add_executable(phase_observation_test test/phase_observation_test.cpp)
    target_link_libraries(phase_observation_test PRIVATE command_lease GTest::gtest_main)
    gtest_discover_tests(phase_observation_test)
    add_executable(navigation_heading_test test/navigation_heading_test.cpp)
    target_link_libraries(navigation_heading_test PRIVATE command_lease GTest::gtest_main)
    gtest_discover_tests(navigation_heading_test)
    add_executable(neutral_release_test test/neutral_release_test.cpp)
    target_link_libraries(neutral_release_test PRIVATE command_lease GTest::gtest_main)
    gtest_discover_tests(neutral_release_test)
    add_executable(owned_mode_contract_test test/owned_mode_contract_test.cpp)
    target_link_libraries(owned_mode_contract_test PRIVATE command_lease nlohmann_json::nlohmann_json GTest::gtest_main)
    target_compile_definitions(owned_mode_contract_test PRIVATE
        PIXEAGLE_OWNERSHIP_PROTOCOL_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/ownership-protocol.json")
    gtest_discover_tests(owned_mode_contract_test)
